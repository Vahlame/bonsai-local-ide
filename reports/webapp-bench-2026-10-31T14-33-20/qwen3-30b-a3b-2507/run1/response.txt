import { createServer } from 'http';
import { readFile, writeFile, stat } from 'fs';
import { join, dirname } from 'path';
import { fileURLToPath } from 'url';

const __filename = fileURLToPath(import.meta.url);
const __dirname = dirname(__filename);

const PORT = process.env.PORT || 3000;
const DATA_FILE = join(__dirname, 'data.json');

const serveStatic = (path, res) => {
  const filePath = join(__dirname, 'public', path);
  stat(filePath, (err, stats) => {
    if (err) {
      res.writeHead(404);
      res.end('Not found');
      return;
    }
    readFile(filePath, (err, data) => {
      if (err) {
        res.writeHead(500);
        res.end('Internal server error');
        return;
      }
      res.writeHead(200, { 'Content-Type': 'text/html' });
      res.end(data);
    });
  });
};

const readTasks = () => {
  return new Promise((resolve, reject) => {
    readFile(DATA_FILE, 'utf8', (err, data) => {
      if (err) {
        if (err.code === 'ENOENT') {
          resolve([]);
        } else {
          reject(err);
        }
      } else {
        try {
          resolve(JSON.parse(data));
        } catch (parseErr) {
          reject(parseErr);
        }
      }
    });
  });
};

const writeTasks = (tasks) => {
  return new Promise((resolve, reject) => {
    writeFile(DATA_FILE, JSON.stringify(tasks, null, 2), 'utf8', (err) => {
      if (err) {
        reject(err);
      } else {
        resolve();
      }
    });
  });
};

const getStats = async () => {
  const tasks = await readTasks();
  const stats = { todo: 0, doing: 0, done: 0, total: 0 };
  tasks.forEach(t => {
    if (t.status === 'todo') stats.todo++;
    else if (t.status === 'doing') stats.doing++;
    else if (t.status === 'done') stats.done++;
    stats.total++;
  });
  return stats;
};

const server = createServer(async (req, res) => {
  const url = new URL(req.url, `http://localhost:${PORT}`);
  const path = url.pathname;

  if (req.method === 'GET' && path === '/') {
    serveStatic('index.html', res);
    return;
  }

  if (req.method === 'GET' && path === '/api/tasks') {
    try {
      const tasks = await readTasks();
      res.writeHead(200, { 'Content-Type': 'application/json' });
      res.end(JSON.stringify(tasks));
    } catch (err) {
      res.writeHead(500);
      res.end('Internal server error');
    }
    return;
  }

  if (req.method === 'POST' && path === '/api/tasks') {
    let body = '';
    req.on('data', chunk => body += chunk);
    req.on('end', async () => {
      try {
        const { title, status = 'todo', priority = 'medium' } = JSON.parse(body);
        if (!title) {
          res.writeHead(400);
          res.end('Title is required');
          return;
        }
        const tasks = await readTasks();
        const newTask = {
          id: Date.now().toString(),
          title,
          status,
          priority,
          createdAt: new Date().toISOString()
        };
        tasks.push(newTask);
        await writeTasks(tasks);
        res.writeHead(201, { 'Content-Type': 'application/json' });
        res.end(JSON.stringify(newTask));
      } catch (err) {
        res.writeHead(400);
        res.end('Invalid request body');
      }
    });
    return;
  }

  if (req.method === 'PATCH' && path.startsWith('/api/tasks/')) {
    const id = path.slice(13);
    let body = '';
    req.on('data', chunk => body += chunk);
    req.on('end', async () => {
      try {
        const updates = JSON.parse(body);
        const tasks = await readTasks();
        const taskIndex = tasks.findIndex(t => t.id === id);
        if (taskIndex === -1) {
          res.writeHead(404);
          res.end('Task not found');
          return;
        }
        const task = tasks[taskIndex];
        const updatedTask = { ...task, ...updates };
        if (updatedTask.status && !['todo', 'doing', 'done'].includes(updatedTask.status)) {
          res.writeHead(400);
          res.end('Invalid status');
          return;
        }
        tasks[taskIndex] = updatedTask;
        await writeTasks(tasks);
        res.writeHead(200, { 'Content-Type': 'application/json' });
        res.end(JSON.stringify(updatedTask));
      } catch (err) {
        res.writeHead(400);
        res.end('Invalid request body');
      }
    });
    return;
  }

  if (req.method === 'DELETE' && path.startsWith('/api/tasks/')) {
    const id = path.slice(13);
    try {
      const tasks = await readTasks();
      const taskIndex = tasks.findIndex(t => t.id === id);
      if (taskIndex === -1) {
        res.writeHead(404);
        res.end('Task not found');
        return;
      }
      tasks.splice(taskIndex, 1);
      await writeTasks(tasks);
      res.writeHead(204);
      res.end();
    } catch (err) {
      res.writeHead(500);
      res.end('Internal server error');
    }
    return;
  }

  if (req.method === 'GET' && path === '/api/stats') {
    try {
      const stats = await getStats();
      res.writeHead(200, { 'Content-Type': 'application/json' });
      res.end(JSON.stringify(stats));
    } catch (err) {
      res.writeHead(500);
      res.end('Internal server error');
    }
    return;
  }

  res.writeHead(404);
  res.end('Not found');
});

server.listen(PORT, '127.0.0.1', () => {
  console.log(`Server running on http://127.0.0.1:${PORT}`);
});

const API_BASE = '/api';

const taskInput = document.getElementById('taskInput');
const addTaskBtn = document.getElementById('addTaskBtn');
const todoList = document.getElementById('todoList');
const doingList = document.getElementById('doingList');
const doneList = document.getElementById('doneList');
const todoCount = document.getElementById('todoCount');
const doingCount = document.getElementById('doingCount');
const doneCount = document.getElementById('doneCount');
const totalCount = document.getElementById('totalCount');

const statusMap = {
  todo: todoList,
  doing: doingList,
  done: doneList
};

const loadTasks = async () => {
  try {
    const response = await fetch(`${API_BASE}/tasks`);
    const tasks = await response.json();
    renderTasks(tasks);
  } catch (err) {
    console.error('Failed to load tasks:', err);
  }
};

const renderTasks = (tasks) => {
  const todo = tasks.filter(t => t.status === 'todo');
  const doing = tasks.filter(t => t.status === 'doing');
  const done = tasks.filter(t => t.status === 'done');

  todoList.innerHTML = '';
  doingList.innerHTML = '';
  doneList.innerHTML = '';

  todo.forEach(task => addTaskToColumn(task, todoList));
  doing.forEach(task => addTaskToColumn(task, doingList));
  done.forEach(task => addTaskToColumn(task, doneList));

  updateStats(todo.length, doing.length, done.length, tasks.length);
};

const addTaskToColumn = (task, column) => {
  const li = document.createElement('li');
  li.setAttribute('data-id', task.id);
  li.innerHTML = `
    <span class="task-title">${task.title}</span>
    <div class="controls">
      <button class="move-btn" data-status="doing">Move to Doing</button>
      <button class="move-btn" data-status="done">Move to Done</button>
      <button class="delete-btn">Delete</button>
    </div>
  `;
  column.appendChild(li);

  const moveBtns = li.querySelectorAll('.move-btn');
  moveBtns.forEach(btn => {
    btn.addEventListener('click', async () => {
      const newStatus = btn.getAttribute('data-status');
      await updateTask(task.id, { status: newStatus });
      loadTasks();
    });
  });

  const deleteBtn = li.querySelector('.delete-btn');
  deleteBtn.addEventListener('click', async () => {
    await deleteTask(task.id);
    loadTasks();
  });
};

const updateTask = async (id, updates) => {
  try {
    const response = await fetch(`${API_BASE}/tasks/${id}`, {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(updates)
    });
    if (!response.ok) {
      throw new Error('Failed to update task');
    }
    return await response.json();
  } catch (err) {
    console.error('Failed to update task:', err);
  }
};

const deleteTask = async (id) => {
  try {
    const response = await fetch(`${API_BASE}/tasks/${id}`, {
      method: 'DELETE'
    });
    if (!response.ok) {
      throw new Error('Failed to delete task');
    }
  } catch (err) {
    console.error('Failed to delete task:', err);
  }
};

const addTask = async () => {
  const title = taskInput.value.trim();
  if (!title) return;
  try {
    const response = await fetch(`${API_BASE}/tasks`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ title })
    });
    if (!response.ok) {
      throw new Error('Failed to add task');
    }
    const newTask = await response.json();
    loadTasks();
    taskInput.value = '';
  } catch (err) {
    console.error('Failed to add task:', err);
  }
};

const updateStats = (todo, doing, done, total) => {
  todoCount.textContent = todo;
  doingCount.textContent = doing;
  doneCount.textContent = done;
  totalCount.textContent = total;
};

addTaskBtn.addEventListener('click', addTask);
taskInput.addEventListener('keypress', (e) => {
  if (e.key === 'Enter') addTask();
});

window.addEventListener('load', loadTasks);
