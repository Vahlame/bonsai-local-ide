A min-heap with FIFO tie-breaking. I'll implement a binary min-heap where the comparison is (priority, insertionOrder). Since insertion order is monotonically increasing, using a counter for insertion order ensures FIFO among equal priorities.

```js
export class PriorityQueue {
  #heap = [];
  #counter = 0;

  enqueue(item, priority) {
    this.#heap.push({ item, priority, order: this.#counter++ });
    this.#siftUp(this.#heap.length - 1);
  }

  get size() {
    return this.#heap.length;
  }

  dequeue() {
    if (this.#heap.length === 0) {
      throw new Error('PriorityQueue is empty');
    }
    const [root] = this.#heap;
    const item = root.item;
    const last = this.#heap.pop();
    if (this.#heap.length > 0) {
      this.#heap[0] = last;
      this.#siftDown(0);
    }
    return item;
  }

  #siftUp(start) {
    const heap = this.#heap;
    let i = start;
    while (i > 0) {
      const parent = (i - 1) >> 1;
      if (heap[i].priority < heap[parent].priority ||
          (heap[i].priority === heap[parent].priority &&
           heap[i].order < heap[parent].order)) {
        [heap[i], heap[parent]] = [heap[parent], heap[i]];
        i = parent;
      } else {
        break;
      }
    }
  }

  #siftDown(start) {
    const heap = this.#heap;
    let i = start;
    const length = heap.length;
    while (true) {
      let left = 2 * i + 1;
      let right = 2 * i + 2;
      let smallest = i;
      if (left < length && this.#less(heap[left], heap[smallest])) {
        smallest = left;
      }
      if (right < length && this.#less(heap[right], heap[smallest])) {
        smallest = right;
      }
      if (smallest === i) break;
      [heap[i], heap[smallest]] = [heap[smallest], heap[i]];
      i = smallest;
    }
  }

  #less(a, b) {
    if (a.priority !== b.priority) return a.priority < b.priority;
    return a.order < b.order;
  }
}
```
