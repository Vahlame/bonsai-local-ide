  #heap = [];

    this.#heap.push({ item, priority });
    const start = this.#heap.length - 1;
    this.#bubbleUp(start);
  }

  dequeue() {
    if (this.#heap.length === 0) {
      throw new Error('PriorityQueue is empty');
    }
    const [first] = this.#heap;
    const last = this.#heap.pop();
    if (this.#heap.length > 0) {
      this.#heap[0] = last;
      this.#bubbleDown(0);
    }
    return last.item;
  }

  get size() {
    return this.#heap.length;
  }

  #bubbleUp(start) {
    let i = start;
    const value = this.#heap[i];
    while (i > 0) {
      const parent = (i - 1) >> 1;
      if (value.priority < this.#heap[parent].priority) {
        this.#heap[i] = this.#heap[parent];
        i = parent;
      } else {
        break;
      }
    }
    this.#heap[i] = value;
  }

  #bubbleDown(start) {
    const length = this.#heap.length;
    const value = this.#heap[start];
    while (true) {
      const left = (start << 1) + 1;
      const right = left + 1;
      let swap = -1;
      if (left < length && value.priority > this.#heap[left].priority) {
        swap = left;
      }
      if (right < length && (swap === -1 || value.priority > this.#heap[right].priority)) {
        swap = right;
      }
      if (swap === -1) break;
      this.#heap[start] = this.#heap[swap];
      start = swap;
    }
    this.#heap[start] = value;
  }

Actually, with a min-heap where we compare `value.priority > this.#heap[left].priority`, we only swap when strictly greater. So equal priorities won't be swapped, which means... hmm, this doesn't necessarily preserve insertion order for equal priorities.

Let me think about this more carefully. The requirement is: on tie (equal priority), the one that came first (FIFO) should come out first.

A standard binary heap doesn't guarantee FIFO ordering for equal keys. To get strict FIFO among equal priorities, I need to incorporate the insertion order into the comparison. I can add a sequence number (insertion index) to each element and compare by (priority, sequence).

Let me use a monotonic counter for insertion order.

```js
export class PriorityQueue {
  #heap = [];
  #counter = 0;

  enqueue(item, priority) {
    this.#heap.push({ item, priority, seq: this.#counter++ });
    const start = this.#heap.length - 1;
    this.#bubbleUp(start);
  }

  dequeue() {
    if (this.#heap.length === 0) {
      throw new Error('PriorityQueue is empty');
    }
    const [first] = this.#heap;
    const last = this.#heap.pop();
    if (this.#heap.length > 0) {
      this.#heap[0] = last;
      this.#bubbleDown(0);
    }
    return last.item;
  }

  get size() {
    return this.#heap.length;
  }

  #bubbleUp(start) {
    let i = start;
    const value = this.#heap[i];
    while (i > 0) {
      const parent = (i - 1) >> 1;
      if (this.#compare(value, this.#heap[parent]) < 0) {
        this.#heap[i] = this.#heap[parent];
        i = parent;
      } else {
        break;
      }
    }
    this.#heap[i] = value;
  }

  #bubbleDown(start) {
    const length = this.#heap.length;
    const value = this.#heap[start];
    while (true) {
      const left = (start << 1) + 1;
      const right = left + 1;
      let swap = -1;
      if (left < length && this.#compare(value, this.#heap[left]) > 0) {
        swap = left;
      }
      if (right < length && this.#compare(value, this.#heap[right]) > 0) {
        swap = right;
      }
      if (swap === -1) break;
      this.#heap[start] = this.#heap[swap];
      start = swap;
    }
    this.#heap[start] = value;
  }

  #compare(a, b) {
    if (a.priority !== b.priority) return a.priority - b.priority;
    return a.seq - b.seq;
  }
}
```

This uses a comparator that first compares priority, then sequence (insertion order). Since seq is unique and increasing, this guarantees FIFO on ties.

Note: comparing priorities with `a.priority - b.priority` works for numbers. If priorities are non-numbers this could break, but the spec implies numeric priorities. I'll keep it simple but robust. Actually, to be safe with the "compare < 0 / > 0" logic, using subtraction is fine for numbers.

Actually, let me reconsider. Using `a.priority - b.priority` returns 0 for equal, negative if a < b, positive if a > b. That's correct for number comparison. Good.
