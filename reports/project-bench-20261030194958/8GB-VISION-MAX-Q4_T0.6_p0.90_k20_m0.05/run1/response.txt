===FILE: src/priority-queue.mjs===
export class PriorityQueue {
  #heap = [];
  #counter = 0;

  enqueue(item, priority) {
    this.#heap.push({ item, priority, seq: this.#counter++ });
    const start = this.#heap.length - 1;
    this.#bubbleUp(start);
  }

  dequeue() {
    if (this.#heap.length === 0) {
      throw new Error('PriorityQueue is empty');
    }
    const [first] = this.#heap;
    const last = this.#heap.pop();
    if (this.#heap.length > 0) {
      this.#heap[0] = last;
      this.#bubbleDown(0);
    }
    return last.item;
  }

  get size() {
    return this.#heap.length;
  }

  #bubbleUp(start) {
    let i = start;
    const value = this.#heap[i];
    while (i > 0) {
      const parent = (i - 1) >> 1;
      if (this.#compare(value, this.#heap[parent]) < 0) {
        this.#heap[i] = this.#heap[parent];
        i = parent;
      } else {
        break;
      }
    }
    this.#heap[i] = value;
  }

  #bubbleDown(start) {
    const length = this.#heap.length;
    const value = this.#heap[start];
    while (true) {
      const left = (start << 1) + 1;
      const right = left + 1;
      let swap = -1;
      if (left < length && this.#compare(value, this.#heap[left]) > 0) {
        swap = left;
      }
      if (right < length && this.#compare(value, this.#heap[right]) > 0) {
        swap = right;
      }
      if (swap === -1) break;
      this.#heap[start] = this.#heap[swap];
      start = swap;
    }
    this.#heap[start] = value;
  }

  #compare(a, b) {
    if (a.priority !== b.priority) return a.priority - b.priority;
    return a.seq - b.seq;
  }
}

import { PriorityQueue } from './priority-queue.mjs';

export class Scheduler {
  constructor({ concurrency = 1 } = {}) {
    if (!Number.isFinite(concurrency) || concurrency < 1) {
      throw new Error('concurrency must be a positive number');
    }
    this.concurrency = concurrency;
    this.queue = new PriorityQueue();
    this.results = [];
    this._ids = 0;
    this._active = 0;
    this._finished = false;
    this._onComplete = null;
  }

  add(fn, { priority = 0, retries = 0 } = {}) {
    const id = this._ids++;
    this.queue.enqueue({ id, fn, priority, retries }, priority);
    return id;
  }

  onResult(handler) {
    this._onComplete = handler;
  }

  async run() {
    this._finished = false;
    this.results = [];
    this._active = 0;

    const slots = new Array(this.concurrency).fill(null);

    const tryStart = () => {
      while (this._active < this.concurrency) {
        const next = this.queue.dequeue();
        if (!next) break;
        slots[this._active] = next;
        this._active++;
      }
      if (this._active === 0 && this.queue.size === 0) {
        this._finished = true;
      }
    };

    tryStart();

    while (this._active > 0) {
      await new Promise((resolve) => {
        const i = this._active - 1;
        const slot = slots[i];

        const runTask = async () => {
          let attempts = 0;
          let lastError;
          let ok = false;
          let value;

          while (attempts <= slot.retries) {
            try {
              value = await slot.fn();
              ok = true;
              break;
            } catch (err) {
              lastError = err;
              attempts++;
              if (attempts <= slot.retries) {
                await new Promise((r) => setTimeout(r, 0));
              }
            }
          }

          if (ok) {
            this.results.push({ id: slot.id, ok: true, value, attempts });
          } else {
            this.results.push({ id: slot.id, ok: false, error: lastError, attempts });
          }

          this._active--;
          slots[i] = null;

          if (this._onComplete) {
            this._onComplete(this.results[this.results.length - 1]);
          }
          resolve();
        };

        runTask();
      });

      tryStart();
    }

    return this.results;
  }
}

import { test } from 'node:test';
import assert from 'node:assert/strict';
import { PriorityQueue } from '../src/priority-queue.mjs';

test('dequeue returns items in priority order', () => {
  const pq = new PriorityQueue();
  pq.enqueue('low', 10);
  pq.enqueue('high', 1);
  pq.enqueue('mid', 5);

  assert.equal(pq.dequeue(), 'high');
  assert.equal(pq.dequeue(), 'mid');
  assert.equal(pq.dequeue(), 'low');
});

test('FIFO on equal priority', () => {
  const pq = new PriorityQueue();
  pq.enqueue('first', 5);
  pq.enqueue('second', 5);
  pq.enqueue('third', 5);

  assert.equal(pq.dequeue(), 'first');
  assert.equal(pq.dequeue(), 'second');
  assert.equal(pq.dequeue(), 'third');
});

test('size reflects number of enqueued items', () => {
  const pq = new PriorityQueue();
  assert.equal(pq.size, 0);
  pq.enqueue('a', 1);
  pq.enqueue('b', 2);
  assert.equal(pq.size, 2);
  pq.dequeue();
  assert.equal(pq.size, 1);
});

test('dequeue on empty throws Error', () => {
  const pq = new PriorityQueue();
  assert.throws(() => pq.dequeue(), Error);
});

import { test } from 'node:test';
import assert from 'node:assert/strict';
import { Scheduler } from '../src/scheduler.mjs';

test('runs tasks respecting priority', async () => {
  const scheduler = new Scheduler({ concurrency: 2 });
  const order = [];
  scheduler.add(async () => { order.push('low'); }, { priority: 10 });
  scheduler.add(async () => { order.push('high'); }, { priority: 1 });
  scheduler.add(async () => { order.push('mid'); }, { priority: 5 });

  const results = await scheduler.run();

  assert.deepEqual(order, ['high', 'mid', 'low']);
  assert.equal(results.length, 3);
  assert.equal(results[0].ok, true);
  assert.equal(results[0].value, 'high');
});

test('respects concurrency limit by measuring max simultaneous', async () => {
  const maxConcurrency = 3;
  const scheduler = new Scheduler({ concurrency: maxConcurrency });
  let maxSimultaneous = 0;
  const active = new Set();

  const tasks = [];
  for (let i = 0; i < 6; i++) {
    tasks.push(scheduler.add(async () => {
      active.add(i);
      maxSimultaneous = Math.max(maxSimultaneous, active.size);
      await new Promise((r) => setTimeout(r, 10));
      active.delete(i);
      return i;
    }, { priority: 0 }));
  }

  await scheduler.run();

  assert.equal(maxSimultaneous, maxConcurrency);
  assert.equal(tasks.length, 6);
});

test('retries on failure then succeeds', async () => {
  const scheduler = new Scheduler({ concurrency: 1 });
  let attempts = 0;
  const task = scheduler.add(async () => {
    attempts++;
    if (attempts < 3) {
      throw new Error('temporary failure');
    }
    return 'ok';
  }, { priority: 0, retries: 5 });

  const results = await scheduler.run();

  assert.equal(results.length, 1);
  assert.equal(results[0].ok, true);
  assert.equal(results[0].value, 'ok');
  assert.equal(results[0].attempts, 3);
});

test('definitive failure returns ok:false with error', async () => {
  const scheduler = new Scheduler({ concurrency: 1 });
  const task = scheduler.add(async () => {
    throw new Error('permanent failure');
  }, { priority: 0, retries: 2 });

  const results = await scheduler.run();

  assert.equal(results.length, 1);
  assert.equal(results[0].ok, false);
  assert.equal(results[0].error.message, 'permanent failure');
  assert.equal(results[0].attempts, 3);
});

test('results are in order of completion by duration', async () => {
  const scheduler = new Scheduler({ concurrency: 1 });
  const ids = [];
  scheduler.add(async () => {
    await new Promise((r) => setTimeout(r, 30));
    return 'slow';
  }, { priority: 0 });
  scheduler.add(async () => {
    await new Promise((r) => setTimeout(r, 10));
    return 'fast';
  }, { priority: 0 });
  scheduler.add(async () => {
    await new Promise((r) => setTimeout(r, 20));
    return 'mid';
  }, { priority: 0 });

  const results = await scheduler.run();

  assert.deepEqual(results.map((r) => r.value), ['fast', 'mid', 'slow']);
});
