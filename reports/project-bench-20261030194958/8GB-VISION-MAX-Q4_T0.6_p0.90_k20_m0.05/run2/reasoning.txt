```js
export class PriorityQueue {
  constructor() {
    this.heap = [];
    this.counter = 0;
  }

  enqueue(item, priority) {
    this.counter++;
    const node = { item, priority, seq: this.counter };
    this.heap.push(node);
    this._bubbleUp(this.heap.length - 1);
  }

  dequeue() {
    if (this.heap.length === 0) {
      throw new Error('PriorityQueue is empty');
    }
    const [first] = this.heap;
    const last = this.heap.pop();
    if (this.heap.length > 0) {
      this.heap[0] = last;
      this._bubbleDown(0);
    }
    return last.item;
  }

  get size() {
    return this.heap.length;
  }

  _bubbleUp(i) {
    const node = this.heap[i];
    while (i > 0) {
      const parent = Math.floor((i - 1) / 2);
      if (this._less(node, this.heap[parent])) {
        [this.heap[i], this.heap[parent]] = [this.heap[parent], this.heap[i]];
        i = parent;
      } else {
        break;
      }
    }
  }

  _bubbleDown(i) {
    const n = this.heap.length;
    while (true) {
      let left = 2 * i + 1;
      let right = 2 * i + 2;
      let smallest = i;
      if (left < n && this._less(this.heap[left], this.heap[smallest])) {
        smallest = left;
      }
      if (right < n && this._less(this.heap[right], this.heap[smallest])) {
        smallest = right;
      }
      if (smallest !== i) {
        [this.heap[i], this.heap[smallest]] = [this.heap[smallest], this.heap[i]];
        i = smallest;
      } else {
        break;
      }
    }
  }

  _less(a, b) {
    if (a.priority !== b.priority) {
      return a.priority < b.priority;
    }
    return a.seq < b.seq;
  }
}
```

This is a proper min-heap. The `_less` function compares by priority first, then by sequence (FIFO).

export class Scheduler {
  constructor({ concurrency = 1 } = {}) {
    this.concurrency = concurrency;
    this.queue = new PriorityQueue();
    this.running = 0;
    this.results = [];
    this.active = new Set();
    this.nextId = 0;
    this._started = false;
  }

  add(fn, { priority = 0, retries = 0 } = {}) {
    const id = ++this.nextId;
    this.queue.enqueue({ id, fn, priority, retries }, priority);
    return id;
  }

  async run() {
    while (this.queue.size > 0 || this.running > 0) {
      // Start tasks up to concurrency limit
      while (this.running < this.concurrency && this.queue.size > 0) {
        const { id, fn, priority, retries } = this.queue.dequeue();
        this.running++;
        this.active.add(id);
        void this._runTask(id, fn, retries).finally(() => {
          this.running--;
          this.active.delete(id);
        });
      }
      // Wait a tick if there's still work
      if (this.queue.size > 0 || this.running > 0) {
        await new Promise((resolve) => setTimeout(resolve, 0));
      }
    }
    return this.results;
  }

  async _runTask(id, fn, retries) {
    let attempts = 0;
    let lastError;
    let value;
    let ok = false;
    while (attempts <= retries) {
      try {
        value = await fn();
        ok = true;
        break;
      } catch (err) {
        lastError = err;
        attempts++;
        if (attempts <= retries) {
          await new Promise((resolve) => setTimeout(resolve, 0));
        }
      }
    }
    const result = { id, ok, attempts };
    if (ok) {
      result.value = value;
    } else {
      result.error = lastError;
    }
    this.results.push(result);
  }
}

test('respects concurrency limit', async () => {
  let active = 0;
  let maxActive = 0;
  const scheduler = new Scheduler({ concurrency: 2 });
  for (let i = 0; i < 10; i++) {
    scheduler.add(async () => {
      active++;
      maxActive = Math.max(maxActive, active);
      await new Promise((resolve) => setTimeout(resolve, 10));
      active--;
    });
  }
  const results = await scheduler.run();
  assert.equal(maxActive, 2);
