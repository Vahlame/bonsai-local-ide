===FILE: src/priority-queue.mjs===
export class PriorityQueue {
  /**
   * Binary min-heap backed priority queue.
   * Ordering: smallest priority first; ties broken by arrival order (FIFO).
   */
  constructor() {
    this.heap = [];
    this.seq = 0;
  }

  enqueue(item, priority) {
    this.seq += 1;
    const node = { item, priority, seq: this.seq };
    this.heap.push(node);
    this._bubbleUp(this.heap.length - 1);
  }

  dequeue() {
    if (this.heap.length === 0) {
      throw new Error('PriorityQueue is empty');
    }
    const first = this.heap[0];
    const last = this.heap.pop();
    if (this.heap.length > 0) {
      this.heap[0] = last;
      this._bubbleDown(0);
    }
    return first.item;
  }

  get size() {
    return this.heap.length;
  }

  _less(a, b) {
    if (a.priority !== b.priority) {
      return a.priority < b.priority;
    }
    return a.seq < b.seq;
  }

  _bubbleUp(i) {
    const node = this.heap[i];
    while (i > 0) {
      const parent = (i - 1) >> 1;
      if (this._less(node, this.heap[parent])) {
        [this.heap[i], this.heap[parent]] = [this.heap[parent], this.heap[i]];
        i = parent;
      } else {
        break;
      }
    }
  }

  _bubbleDown(i) {
    const n = this.heap.length;
    while (true) {
      let smallest = i;
      const left = 2 * i + 1;
      const right = 2 * i + 2;
      if (left < n && this._less(this.heap[left], this.heap[smallest])) {
        smallest = left;
      }
      if (right < n && this._less(this.heap[right], this.heap[smallest])) {
        smallest = right;
      }
      if (smallest !== i) {
        [this.heap[i], this.heap[smallest]] = [this.heap[smallest], this.heap[i]];
        i = smallest;
      } else {
        break;
      }
    }
  }
}

===FILE: src/scheduler.mjs===
import { PriorityQueue } from './priority-queue.mjs';

export class Scheduler {
  /**
   * Runs async tasks respecting priority, a concurrency ceiling, and a retry
   * budget. Results are returned in completion order.
   */
  constructor({ concurrency = 1 } = {}) {
    this.concurrency = concurrency;
    this.queue = new PriorityQueue();
    this.running = 0;
    this.results = [];
    this.active = new Set();
    this.nextId = 0;
    this._started = false;
  }

  add(fn, { priority = 0, retries = 0 } = {}) {
    const id = ++this.nextId;
    this.queue.enqueue({ id, fn, priority, retries }, priority);
    return id;
  }

  async run() {
    if (this._started) {
      throw new Error('Scheduler.run() can only be called once');
    }
    this._started = true;

    while (this.queue.size > 0 || this.running > 0) {
      while (this.running < this.concurrency && this.queue.size > 0) {
        const { id, fn, priority, retries } = this.queue.dequeue();
        this.running += 1;
        this.active.add(id);
        void this._runTask(id, fn, retries).finally(() => {
          this.running -= 1;
          this.active.delete(id);
        });
      }
      if (this.queue.size > 0 || this.running > 0) {
        await new Promise((resolve) => setTimeout(resolve, 0));
      }
    }
    return this.results;
  }

  async _runTask(id, fn, retries) {
    let attempts = 0;
    let value;
    let ok = false;
    let lastError;

    while (attempts <= retries) {
      try {
        value = await fn();
        ok = true;
        break;
      } catch (err) {
        lastError = err;
        attempts += 1;
        if (attempts <= retries) {
          await new Promise((resolve) => setTimeout(resolve, 0));
        }
      }
    }

    const result = { id, ok, attempts };
    if (ok) {
      result.value = value;
    } else {
      result.error = lastError;
    }
    this.results.push(result);
  }
}

===FILE: test/priority-queue.test.mjs===
import { test } from 'node:test';
import assert from 'node:assert/strict';
import { PriorityQueue } from '../src/priority-queue.mjs';

test('orders by priority (min-heap)', () => {
  const pq = new PriorityQueue();
  pq.enqueue('c', 3);
  pq.enqueue('a', 1);
  pq.enqueue('b', 2);

  assert.equal(pq.dequeue(), 'a');
  assert.equal(pq.dequeue(), 'b');
  assert.equal(pq.dequeue(), 'c');
});

test('breaks ties FIFO by arrival order', () => {
  const pq = new PriorityQueue();
  pq.enqueue('first', 5);
  pq.enqueue('second', 5);
  pq.enqueue('third', 5);

  assert.equal(pq.dequeue(), 'first');
  assert.equal(pq.dequeue(), 'second');
  assert.equal(pq.dequeue(), 'third');
});

test('mixes priority ordering with FIFO within same priority', () => {
  const pq = new PriorityQueue();
  pq.enqueue('low-early', 1);
  pq.enqueue('high', 5);
  pq.enqueue('low-late', 1);

  assert.equal(pq.dequeue(), 'low-early');
  assert.equal(pq.dequeue(), 'low-late');
  assert.equal(pq.dequeue(), 'high');
});

test('size reflects contents', () => {
  const pq = new PriorityQueue();
  assert.equal(pq.size, 0);
  pq.enqueue('a', 1);
  pq.enqueue('b', 2);
  assert.equal(pq.size, 2);
  pq.dequeue();
  assert.equal(pq.size, 1);
});

test('dequeue on empty throws Error', () => {
  const pq = new PriorityQueue();
  assert.throws(() => pq.dequeue(), (err) => err instanceof Error);
});

import { test } from 'node:test';
import assert from 'node:assert/strict';
import { Scheduler } from '../src/scheduler.mjs';

test('respects concurrency limit (measures max simultaneous)', async () => {
  let active = 0;
  let maxActive = 0;
  const scheduler = new Scheduler({ concurrency: 2 });
  for (let i = 0; i < 10; i++) {
    scheduler.add(async () => {
      active += 1;
      maxActive = Math.max(maxActive, active);
      await new Promise((resolve) => setTimeout(resolve, 10));
      active -= 1;
    });
  }
  const results = await scheduler.run();
  assert.equal(maxActive, 2);
  assert.equal(results.length, 10);
  assert.ok(results.every((r) => r.ok));
});

test('runs at least one task at a time by default', async () => {
  let active = 0;
  let maxActive = 0;
  const scheduler = new Scheduler({ concurrency: 1 });
  for (let i = 0; i < 3; i++) {
    scheduler.add(async () => {
      active += 1;
      maxActive = Math.max(maxActive, active);
      await new Promise((resolve) => setTimeout(resolve, 5));
      active -= 1;
    });
  }
  await scheduler.run();
  assert.equal(maxActive, 1);
});

test('retries until success after 2 failures', async () => {
  let attempts = 0;
  const scheduler = new Scheduler({ concurrency: 1 });
  scheduler.add(async () => {
    attempts += 1;
    if (attempts < 3) {
      throw new Error('temporary');
    }
    return 'done';
  }, { priority: 0, retries: 2 });

  const results = await scheduler.run();
  assert.equal(results.length, 1);
  const [result] = results;
  assert.equal(result.ok, true);
  assert.equal(result.attempts, 3);
  assert.equal(result.value, 'done');
});

test('reports definitive failure when retries exhausted', async () => {
  const scheduler = new Scheduler({ concurrency: 1 });
  scheduler.add(async () => {
    throw new Error('permanent');
  }, { priority: 0, retries: 2 });

  const results = await scheduler.run();
  const [result] = results;
  assert.equal(result.ok, false);
  assert.equal(result.attempts, 3);
  assert.equal(result.error.message, 'permanent');
  assert.equal(result.value, undefined);
});

test('retries=0 runs fn exactly once and surfaces the failure', async () => {
  const scheduler = new Scheduler({ concurrency: 1 });
  scheduler.add(async () => {
    throw new Error('once');
  }, { priority: 0, retries: 0 });

  const results = await scheduler.run();
  const [result] = results;
  assert.equal(result.ok, false);
  assert.equal(result.attempts, 1);
  assert.equal(result.error.message, 'once');
});

test('returns results in completion order by duration', async () => {
  const order = [];
  const scheduler = new Scheduler({ concurrency: 4 });
  scheduler.add(async () => {
    await new Promise((resolve) => setTimeout(() => order.push('long'), 100));
  }, { priority: 0, retries: 0 });
  scheduler.add(async () => {
    await new Promise((resolve) => setTimeout(() => order.push('short'), 10));
  }, { priority: 0, retries: 0 });
  scheduler.add(async () => {
    await new Promise((resolve) => setTimeout(() => order.push('mid'), 50));
  }, { priority: 0, retries: 0 });

  const results = await scheduler.run();
  assert.deepEqual(order, ['short', 'mid', 'long']);
  // Completion order matches the pushed order.
  assert.deepEqual(results.map((r) => r.id), [2, 3, 1]);
});

test('priority ordering is honored across concurrent runs', async () => {
  const results = await new Scheduler({ concurrency: 4 }).run();
  assert.deepEqual(results.map((r) => r.id), [1, 2, 3, 4]);
});
