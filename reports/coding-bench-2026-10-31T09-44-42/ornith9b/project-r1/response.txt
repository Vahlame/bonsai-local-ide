export class PriorityQueue {
  constructor() {
    this._heap = [];
    this._seq = 0;
  }

  enqueue(item, priority) {
    const node = { item, priority, seq: this._seq++ };
    this._heap.push(node);
    this._siftUp(this._heap.length - 1);
  }

  dequeue() {
    if (this._heap.length === 0) {
      throw new Error('PriorityQueue is empty');
    }
    const root = this._heap[0];
    const last = this._heap.pop();
    if (this._heap.length > 0) {
      this._heap[0] = last;
      this._siftDown(0);
    }
    return last.item;
  }

  get size() {
    return this._heap.length;
  }

  _siftUp(start) {
    const heap = this._heap;
    let i = start;
    while (i > 0) {
      const parent = (i - 1) >> 1;
      if (this._compare(heap[i], heap[parent]) < 0) {
        const tmp = heap[i];
        heap[i] = heap[parent];
        heap[parent] = tmp;
        i = parent;
      } else {
        break;
      }
    }
  }

  _siftDown(start) {
    const heap = this._heap;
    const length = heap.length;
    let i = start;
    while (true) {
      let left = 2 * i + 1;
      let right = 2 * i + 2;
      let smallest = i;
      if (left < length && this._compare(heap[left], heap[smallest]) < 0) {
        smallest = left;
      }
      if (right < length && this._compare(heap[right], heap[smallest]) < 0) {
        smallest = right;
      }
      if (smallest === i) {
        break;
      }
      const tmp = heap[i];
      heap[i] = heap[smallest];
      heap[smallest] = tmp;
      i = smallest;
    }
  }

  _compare(a, b) {
    if (a.priority !== b.priority) {
      return a.priority - b.priority;
    }
    return a.seq - b.seq;
  }
}

import { PriorityQueue } from './priority-queue.mjs';

export class Scheduler {
  constructor({ concurrency = 1 }) {
    if (!Number.isInteger(concurrency) || concurrency < 1) {
      throw new Error('concurrency must be a positive integer');
    }
    this.concurrency = concurrency;
    this.queue = new PriorityQueue();
    this._idCounter = 0;
    this._running = 0;
    this._finished = [];
    this._stopped = false;
  }

  add(fn, { priority = 0, retries = 0 } = {}) {
    const id = this._idCounter++;
    this.queue.enqueue({ id, fn, priority, retries }, priority);
    return id;
  }

  async run() {
    this._stopped = false;
    this._finished = [];
    this._running = 0;

    const workers = [];
    for (let i = 0; i < this.concurrency; i++) {
      workers.push(this._worker());
    }

    await Promise.all(workers);
    return this._finished;
  }

  _worker() {
    return async () => {
      while (true) {
        if (this._stopped) {
          return;
        }
        const task = this.queue.dequeue();
        if (!task) {
          break;
        }

        this._running++;
        try {
          let attempts = 0;
          let ok = false;
          let value;
          let error;
          while (attempts <= task.retries) {
            attempts++;
            try {
              const result = task.fn(task.id);
              value = await result;
              ok = true;
              break;
            } catch (err) {
              error = err;
            }
          }
          this._finished.push({
            id: task.id,
            ok,
            value: ok ? value : undefined,
            error: ok ? undefined : error,
            attempts,
          });
        } catch (err) {
          this._finished.push({
            id: task.id,
            ok: false,
            value: undefined,
            error: err,
            attempts,
          });
        } finally {
          this._running--;
        }
      }
    };
  }
}
