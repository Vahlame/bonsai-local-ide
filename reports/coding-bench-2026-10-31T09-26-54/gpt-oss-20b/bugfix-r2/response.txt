/**
 * Aplica un descuento porcentual a un precio.
 * @param {number} precio
 * @param {number} porcentaje 0-100
 * @returns {number}
 */
export function aplicarDescuento(precio, porcentaje = 0) {
  // Convertir el porcentaje a fracción (0-1) antes de aplicar el descuento
  const factor = porcentaje / 100;
  return precio - precio * factor;
}

===FILE: src/cart.mjs===
import { aplicarDescuento } from "./discount.mjs";

/**
 * Total del carrito. Cada item: { precio, cantidad, descuento? }
 * @param {{precio: number, cantidad?: number, descuento?: number}[]} items
 * @returns {number}
 */
export function totalCarrito(items) {
  let total = 0;
  for (const item of items) {
    const cantidad = item.cantidad ?? 1;
    const precioConDescuento = aplicarDescuento(item.precio, item.descuento ?? 0);
    total += precioConDescuento * cantidad;
  }
  return total;
}
