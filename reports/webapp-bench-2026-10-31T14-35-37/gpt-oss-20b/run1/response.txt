import { createServer } from 'http';
import { readFile, writeFile, access, mkdir } from 'fs/promises';
import { join, extname } from 'path';
import { fileURLToPath } from 'url';

const __dirname = fileURLToPath(new URL('.', import.meta.url));
const DATA_FILE = join(__dirname, 'data.json');
const PUBLIC_DIR = join(__dirname, 'public');
const PORT = process.env.PORT ? parseInt(process.env.PORT, 10) : 3000;

// Simple MIME type mapping
const MIME_TYPES = {
  '.html': 'text/html',
  '.js': 'application/javascript',
  '.css': 'text/css',
  '.json': 'application/json',
  '.png': 'image/png',
  '.jpg': 'image/jpeg',
  '.jpeg': 'image/jpeg',
  '.svg': 'image/svg+xml',
  '.ico': 'image/x-icon',
};

async function ensureDataFile() {
  try {
    await access(DATA_FILE);
  } catch {
    await writeFile(DATA_FILE, '[]', 'utf8');
  }
}

async function loadData() {
  await ensureDataFile();
  const content = await readFile(DATA_FILE, 'utf8');
  try {
    return JSON.parse(content);
  } catch {
    return [];
  }
}

async function saveData(data) {
  await writeFile(DATA_FILE, JSON.stringify(data, null, 2), 'utf8');
}

function getNextId(data) {
  return data.reduce((max, t) => Math.max(max, t.id), 0) + 1;
}

function getStats(data) {
  const stats = { todo: 0, doing: 0, done: 0, total: data.length };
  for (const t of data) {
    if (t.status === 'todo') stats.todo++;
    else if (t.status === 'doing') stats.doing++;
    else if (t.status === 'done') stats.done++;
  }
  return stats;
}

async function parseBody(req) {
  return new Promise((resolve, reject) => {
    let body = '';
    req.on('data', chunk => (body += chunk));
    req.on('end', () => {
      try {
        resolve(JSON.parse(body));
      } catch {
        resolve({});
      }
    });
    req.on('error', reject);
  });
}

async function handleApi(req, res) {
  const url = new URL(req.url, `http://${req.headers.host}`);
  const method = req.method;
  const pathname = url.pathname;

  if (pathname === '/api/tasks' && method === 'GET') {
    const data = await loadData();
    res.writeHead(200, { 'Content-Type': 'application/json' });
    res.end(JSON.stringify(data));
    return;
  }

  if (pathname === '/api/tasks' && method === 'POST') {
    const body = await parseBody(req);
    const title = body.title?.trim();
    if (!title) {
      res.writeHead(400, { 'Content-Type': 'application/json' });
      res.end(JSON.stringify({ error: 'Title is required' }));
      return;
    }
    const data = await loadData();
    const newTask = {
      id: getNextId(data),
      title,
      status: body.status === 'doing' || body.status === 'done' ? body.status : 'todo',
      priority: typeof body.priority === 'number' ? body.priority : 0,
      createdAt: new Date().toISOString(),
    };
    data.push(newTask);
    await saveData(data);
    res.writeHead(201, { 'Content-Type': 'application/json' });
    res.end(JSON.stringify(newTask));
    return;
  }

  const taskIdMatch = pathname.match(/^\/api\/tasks\/(\d+)$/);
  if (taskIdMatch) {
    const id = parseInt(taskIdMatch[1], 10);
    const data = await loadData();
    const index = data.findIndex(t => t.id === id);
    if (index === -1) {
      res.writeHead(404, { 'Content-Type': 'application/json' });
      res.end(JSON.stringify({ error: 'Task not found' }));
      return;
    }

    if (method === 'PATCH') {
      const body = await parseBody(req);
      const task = data[index];
      if (body.title !== undefined) task.title = body.title.trim();
      if (body.status && ['todo', 'doing', 'done'].includes(body.status)) task.status = body.status;
      if (body.priority !== undefined) task.priority = body.priority;
      await saveData(data);
      res.writeHead(200, { 'Content-Type': 'application/json' });
      res.end(JSON.stringify(task));
      return;
    }

    if (method === 'DELETE') {
      data.splice(index, 1);
      await saveData(data);
      res.writeHead(204);
      res.end();
      return;
    }
  }

  if (pathname === '/api/stats' && method === 'GET') {
    const data = await loadData();
    const stats = getStats(data);
    res.writeHead(200, { 'Content-Type': 'application/json' });
    res.end(JSON.stringify(stats));
    return;
  }

  // If no API route matched
  res.writeHead(404, { 'Content-Type': 'application/json' });
  res.end(JSON.stringify({ error: 'Not found' }));
}

async function serveStatic(req, res) {
  let pathname = new URL(req.url, `http://${req.headers.host}`).pathname;
  if (pathname === '/') pathname = '/index.html';
  const filePath = join(PUBLIC_DIR, pathname);
  try {
    const data = await readFile(filePath);
    const ext = extname(filePath);
    const mime = MIME_TYPES[ext] || 'application/octet-stream';
    res.writeHead(200, { 'Content-Type': mime });
    res.end(data);
  } catch {
    res.writeHead(404, { 'Content-Type': 'text/plain' });
    res.end('404 Not Found');
  }
}

const server = createServer(async (req, res) => {
  if (req.url.startsWith('/api/')) {
    await handleApi(req, res);
  } else {
    await serveStatic(req, res);
  }
});

server.listen(PORT, '127.0.0.1', () => {
  console.log(`Server running at http://127.0.0.1:${PORT}/`);
});

===FILE: public/app.js===
const API_BASE = '/api/tasks';

async function fetchTasks() {
  const res = await fetch(API_BASE);
  return res.json();
}

async function fetchStats() {
  const res = await fetch('/api/stats');
  return res.json();
}

function createTaskElement(task) {
  const li = document.createElement('li');
  li.className = 'task';
  li.dataset.id = task.id;
  li.textContent = task.title;
  li.title = `Priority: ${task.priority}\nCreated: ${new Date(task.createdAt).toLocaleString()}`;

  li.addEventListener('click', () => cycleStatus(task.id));
  const delBtn = document.createElement('button');
  delBtn.textContent = '✕';
  delBtn.className = 'delete-btn';
  delBtn.addEventListener('click', (e) => {
    e.stopPropagation();
    deleteTask(task.id);
  });
  li.appendChild(delBtn);
  return li;
}

function cycleStatus(id) {
  // Get current status
  const task = currentTasks.find(t => t.id === id);
  if (!task) return;
  const next = { todo: 'doing', doing: 'done', done: 'todo' }[task.status];
  updateTask(id, { status: next });
}

async function addTask(title) {
  const res = await fetch(API_BASE, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ title })
  });
  if (res.ok) {
    const task = await res.json();
    currentTasks.push(task);
    renderTasks();
    renderStats();
  }
}

async function updateTask(id, updates) {
  const res = await fetch(`${API_BASE}/${id}`, {
    method: 'PATCH',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(updates)
  });
  if (res.ok) {
    const updated = await res.json();
    const idx = currentTasks.findIndex(t => t.id === id);
    if (idx !== -1) currentTasks[idx] = updated;
    renderTasks();
    renderStats();
  }
}

async function deleteTask(id) {
  const res = await fetch(`${API_BASE}/${id}`, { method: 'DELETE' });
  if (res.ok) {
    currentTasks = currentTasks.filter(t => t.id !== id);
    renderTasks();
    renderStats();
  }
}

function renderTasks() {
  const todoList = document.getElementById('todo-list');
  const doingList = document.getElementById('doing-list');
  const doneList = document.getElementById('done-list');
  todoList.innerHTML = '';
  doingList.innerHTML = '';
  doneList.innerHTML = '';
  for (const task of currentTasks) {
    const el = createTaskElement(task);
    if (task.status === 'todo') todoList.appendChild(el);
    else if (task.status === 'doing') doingList.appendChild(el);
    else if (task.status === 'done') doneList.appendChild(el);
  }
}

function renderStats() {
  const statsEl = document.getElementById('stats-content');
  fetchStats().then(stats => {
    statsEl.textContent = `Total: ${stats.total} | Todo: ${stats.todo} | Doing: ${stats.doing} | Done: ${stats.done}`;
  });
}

let currentTasks = [];

document.addEventListener('DOMContentLoaded', async () => {
  currentTasks = await fetchTasks();
  renderTasks();
  renderStats();

  const addBtn = document.getElementById('add-task-btn');
  const input = document.getElementById('new-task-title');
  addBtn.addEventListener('click', () => {
    const title = input.value.trim();
    if (title) {
      addTask(title);
      input.value = '';
    }
  });
});
