import http from 'http';
import fs from 'fs';
import path from 'path';
import { fileURLToPath } from 'url';

const __dirname = path.dirname(fileURLToPath(import.meta.url));
const PORT = process.env.PORT || 3000;
const HOST = '127.0.0.1';
const DATA_FILE = path.join(__dirname, 'data.json');

const MIME_TYPES = {
  '.html': 'text/html',
  '.css': 'text/css',
  '.js': 'application/javascript',
  '.json': 'application/json'
};

function readTasks() {
  try {
    const data = fs.readFileSync(DATA_FILE, 'utf-8');
    return JSON.parse(data);
  } catch (e) {
    return [];
  }
}

function writeTasks(tasks) {
  fs.writeFileSync(DATA_FILE, JSON.stringify(tasks, null, 2), 'utf-8');
}

function createTask(title, status = 'todo', priority = 'medium') {
  return {
    id: crypto.randomUUID(),
    title,
    status,
    priority,
    createdAt: new Date().toISOString()
  };
}

const server = http.createServer((req, res) => {
  const parsedUrl = new URL(req.url, `http://${req.headers.host}`);
  const pathname = parsedUrl.pathname;

  // Serve static files
  if (pathname === '/' || pathname === '/index.html') {
    const filePath = path.join(__dirname, 'public', 'index.html');
    const ext = path.extname(filePath);
    res.writeHead(200, { 'Content-Type': MIME_TYPES[ext] || 'text/plain' });
    fs.createReadStream(filePath).pipe(res);
    return;
  }

  if (pathname.startsWith('/public/')) {
    const filePath = path.join(__dirname, pathname);
    const ext = path.extname(filePath);
    if (fs.existsSync(filePath)) {
      res.writeHead(200, { 'Content-Type': MIME_TYPES[ext] || 'text/plain' });
      fs.createReadStream(filePath).pipe(res);
    } else {
      res.writeHead(404);
      res.end('Not Found');
    }
    return;
  }

  // API Routes
  const apiPrefix = '/api';
  if (pathname.startsWith(apiPrefix)) {
    const apiPath = pathname.slice(apiPrefix.length);

    // GET /api/tasks
    if (req.method === 'GET' && apiPath === '/tasks') {
      const tasks = readTasks();
      res.writeHead(200, { 'Content-Type': 'application/json' });
      res.end(JSON.stringify(tasks));
      return;
    }

    // GET /api/stats
    if (req.method === 'GET' && apiPath === '/stats') {
      const tasks = readTasks();
      const stats = {
        todo: tasks.filter(t => t.status === 'todo').length,
        doing: tasks.filter(t => t.status === 'doing').length,
        done: tasks.filter(t => t.status === 'done').length,
        total: tasks.length
      };
      res.writeHead(200, { 'Content-Type': 'application/json' });
      res.end(JSON.stringify(stats));
      return;
    }

    // POST /api/tasks
    if (req.method === 'POST' && apiPath === '/tasks') {
      let body = '';
      req.on('data', chunk => body += chunk);
      req.on('end', () => {
        try {
          const { title, status, priority } = JSON.parse(body || '{}');
          if (!title) throw new Error('Title is required');
          
          const tasks = readTasks();
          const newTask = createTask(title, status || 'todo', priority || 'medium');
          tasks.push(newTask);
          writeTasks(tasks);
          
          res.writeHead(201, { 'Content-Type': 'application/json' });
          res.end(JSON.stringify(newTask));
        } catch (e) {
          res.writeHead(400, { 'Content-Type': 'application/json' });
          res.end(JSON.stringify({ error: e.message }));
        }
      });
      return;
    }

    // PATCH /api/tasks/:id
    if (req.method === 'PATCH' && apiPath.startsWith('/tasks/')) {
      const id = apiPath.split('/')[2];
      let body = '';
      req.on('data', chunk => body += chunk);
      req.on('end', () => {
        try {
          const updates = JSON.parse(body || '{}');
          const tasks = readTasks();
          const index = tasks.findIndex(t => t.id === id);
          
          if (index === -1) {
            res.writeHead(404);
            res.end('Not Found');
            return;
          }

          // Validate status if provided
          if (updates.status && !['todo', 'doing', 'done'].includes(updates.status)) {
            throw new Error('Invalid status');
          }

          tasks[index] = { ...tasks[index], ...updates };
          writeTasks(tasks);
          
          res.writeHead(200, { 'Content-Type': 'application/json' });
          res.end(JSON.stringify(tasks[index]));
        } catch (e) {
          res.writeHead(400, { 'Content-Type': 'application/json' });
          res.end(JSON.stringify({ error: e.message }));
        }
      });
      return;
    }

    // DELETE /api/tasks/:id
    if (req.method === 'DELETE' && apiPath.startsWith('/tasks/')) {
      const id = apiPath.split('/')[2];
      const tasks = readTasks();
      const newTasks = tasks.filter(t => t.id !== id);
      
      if (tasks.length === newTasks.length) {
        res.writeHead(404);
        res.end('Not Found');
        return;
      }
      
      writeTasks(newTasks);
      res.writeHead(204);
      res.end();
      return;
    }
  }

  // 404 for everything else
  res.writeHead(404);
  res.end('Not Found');
});

server.listen(PORT, HOST, () => {
  console.log(`Server running at http://${HOST}:${PORT}/`);
});

===FILE: public/app.js===
document.addEventListener('DOMContentLoaded', () => {
  const taskInput = document.getElementById('task-input');
  const addBtn = document.getElementById('add-btn');
  const board = document.getElementById('board');
  const statsContainer = document.getElementById('stats');

  // Load initial data
  loadTasks();
  loadStats();

  // Event Listeners
  addBtn.addEventListener('click', addTask);
  taskInput.addEventListener('keypress', (e) => {
    if (e.key === 'Enter') addTask();
  });

  // Delegate events for dynamic elements
  board.addEventListener('click', handleBoardClick);

  async function loadTasks() {
    try {
      const res = await fetch('/api/tasks');
      const tasks = await res.json();
      renderTasks(tasks);
    } catch (e) {
      console.error('Error loading tasks', e);
    }
  }

  async function loadStats() {
    try {
      const res = await fetch('/api/stats');
      const stats = await res.json();
      renderStats(stats);
    } catch (e) {
      console.error('Error loading stats', e);
    }
  }

  function renderTasks(tasks) {
    const lists = {
      todo: document.getElementById('list-todo'),
      doing: document.getElementById('list-doing'),
      done: document.getElementById('list-done')
    };

    // Clear lists
    Object.values(lists).forEach(el => el.innerHTML = '');

    tasks.forEach(task => {
      const card = document.createElement('div');
      card.className = 'task-card';
      card.dataset.id = task.id;
      
      const title = document.createElement('div');
      title.className = 'task-title';
      title.textContent = task.title;
      
      const priority = document.createElement('span');
      priority.className = 'task-priority';
      priority.textContent = `[${task.priority}]`;
      
      const controls = document.createElement('div');
      controls.className = 'task-controls';

      // Add buttons based on status
      if (task.status !== 'doing') {
        const moveBtn = document.createElement('button');
        moveBtn.textContent = '→';
        moveBtn.title = 'Mover a Doing';
        moveBtn.dataset.action = 'move-to-doing';
        controls.appendChild(moveBtn);
      }
      
      if (task.status !== 'done') {
        const moveBtn = document.createElement('button');
        moveBtn.textContent = '✓';
        moveBtn.title = 'Mover a Done';
        moveBtn.dataset.action = 'move-to-done';
        controls.appendChild(moveBtn);
      }

      if (task.status !== 'todo') {
        const moveBtn = document.createElement('button');
        moveBtn.textContent = '←';
        moveBtn.title = 'Mover a Todo';
        moveBtn.dataset.action = 'move-to-todo';
        controls.appendChild(moveBtn);
      }

      const delBtn = document.createElement('button');
      delBtn.textContent = '×';
      delBtn.title = 'Eliminar';
      delBtn.dataset.action = 'delete';
      controls.appendChild(delBtn);

      card.appendChild(title);
      card.appendChild(priority);
      card.appendChild(controls);
      lists[task.status].appendChild(card);
    });
  }

  function renderStats(stats) {
    const spans = statsContainer.querySelectorAll('b');
    spans[0].textContent = stats.total;
    spans[1].textContent = stats.todo;
    spans[2].textContent = stats.doing;
    spans[3].textContent = stats.done;
  }

  async function addTask() {
    const title = taskInput.value.trim();
    if (!title) return;

    try {
      const res = await fetch('/api/tasks', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ title })
      });
      if (res.ok) {
        taskInput.value = '';
        loadTasks();
        loadStats();
      }
    } catch (e) {
      console.error('Error adding task', e);
    }
  }

  async function handleBoardClick(e) {
    const btn = e.target.closest('button');
    if (!btn) return;

    const card = btn.closest('.task-card');
    if (!card) return;

    const id = card.dataset.id;
    const action = btn.dataset.action;

    if (action === 'delete') {
      try {
        await fetch(`/api/tasks/${id}`, { method: 'DELETE' });
        loadTasks();
        loadStats();
      } catch (e) {
        console.error('Error deleting', e);
      }
    } else if (action.startsWith('move-to-')) {
      const newStatus = action.split('-')[2]; // todo, doing, done
      try {
        await fetch(`/api/tasks/${id}`, {
          method: 'PATCH',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ status: newStatus })
        });
        loadTasks();
        loadStats();
      } catch (e) {
        console.error('Error moving task', e);
      }
    }
  }
});
