export class PriorityQueue {
  constructor() {
    this._queue = [];
    this._order = 0;
  }

  enqueue(item, priority) {
    this._queue.push({ item, priority, order: this._order++ });
  }

  dequeue() {
    if (this._queue.length === 0) {
      throw new Error('PriorityQueue is empty');
    }
    let minIndex = 0;
    let min = this._queue[0];
    for (let i = 1; i < this._queue.length; i++) {
      const current = this._queue[i];
      if (
        current.priority < min.priority ||
        (current.priority === min.priority && current.order < min.order)
      ) {
        min = current;
        minIndex = i;
      }
    }
    const [removed] = this._queue.splice(minIndex, 1);
    return removed.item;
  }

  get size() {
    return this._queue.length;
  }

  isEmpty() {
    return this._queue.length === 0;
  }
}

import { PriorityQueue } from './priority-queue.mjs';

export class Scheduler {
  constructor({ concurrency }) {
    if (typeof concurrency !== 'number' || concurrency <= 0) {
      throw new Error('concurrency must be a positive number');
    }
    this.concurrency = concurrency;
    this._queue = new PriorityQueue();
    this._idCounter = 0;
  }

  add(fn, { priority = 0, retries = 0 } = {}) {
    if (typeof fn !== 'function') {
      throw new Error('fn must be a function');
    }
    const id = ++this._idCounter;
    this._queue.enqueue({ id, fn, priority, retries }, priority);
    return id;
  }

  get size() {
    return this._queue.size;
  }

  async run() {
    const results = [];
    const tasks = [];
    while (!this._queue.isEmpty()) {
      tasks.push(this._queue.dequeue());
    }

    const runNext = async () => {
      if (tasks.length === 0) return;
      const task = tasks.shift();
      const res = await this._executeTask(task);
      results.push(res);
      await runNext();
    };

    const workers = [];
    const workerCount = Math.min(this.concurrency, tasks.length);
    for (let i = 0; i < workerCount; i++) {
      workers.push(runNext());
    }

    await Promise.all(workers);
    return results;
  }

  async _executeTask(task) {
    let attempts = 0;
    while (true) {
      attempts++;
      try {
        const value = await task.fn();
        return { id: task.id, ok: true, value, attempts };
      } catch (error) {
        if (attempts <= task.retries) continue;
        return { id: task.id, ok: false, error, attempts };
      }
    }
  }
}
