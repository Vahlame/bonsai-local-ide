== Python ==
import json
from dataclasses import dataclass, field
from pathlib import Path


@dataclass
class Pedido:
    id: int
    cliente: str
    lineas: list[tuple[str, int, float]] = field(default_factory=list)

    def total(self) -> float:
        return round(sum(cantidad * precio for _, cantidad, precio in self.lineas), 2)


def cargar(ruta: Path) -> list[Pedido]:
    datos = json.loads(ruta.read_text(encoding="utf-8"))
    return [Pedido(p["id"], p["cliente"], [tuple(l) for l in p["lineas"]]) for p in datos]


def resumen(pedidos: list[Pedido]) -> dict[str, float]:
    por_cliente: dict[str, float] = {}
    for pedido in pedidos:
        por_cliente[pedido.cliente] = por_cliente.get(pedido.cliente, 0.0) + pedido.total()
    return dict(sorted(por_cliente.items(), key=lambda kv: kv[1], reverse=True))


if __name__ == "__main__":
    for cliente, total in resumen(cargar(Path("pedidos.json"))).items():
        print(f"{cliente:<20} {total:>10.2f}")

export async function descargar(url, { reintentos = 3, esperaMs = 500 } = {}) {
  for (let intento = 1; intento <= reintentos; intento += 1) {
    try {
      const res = await fetch(url, { signal: AbortSignal.timeout(10_000) });
      if (!res.ok) throw new Error(`HTTP ${res.status}`);
      return await res.json();
    } catch (err) {
      if (intento === reintentos) throw err;
      await new Promise((resolve) => setTimeout(resolve, esperaMs * intento));
    }
  }
}

const cache = new Map();
export function memoizar(fn) {
  return (...args) => {
    const clave = JSON.stringify(args);
    if (!cache.has(clave)) cache.set(clave, fn(...args));
    return cache.get(clave);
  };
}

== SQL ==
SELECT c.nombre, COUNT(p.id) AS pedidos, SUM(l.cantidad * l.precio) AS total
FROM clientes c
JOIN pedidos p ON p.cliente_id = c.id
JOIN lineas l ON l.pedido_id = p.id
WHERE p.fecha >= DATE '2026-01-01' AND p.estado <> 'cancelado'
GROUP BY c.nombre
HAVING SUM(l.cantidad * l.precio) > 1000
ORDER BY total DESC
LIMIT 20;

CREATE INDEX idx_pedidos_cliente_fecha ON pedidos (cliente_id, fecha DESC);

typedef struct nodo { int valor; struct nodo *sig; } nodo;

nodo *invertir(nodo *cabeza) {
    nodo *prev = NULL;
    while (cabeza) {
        nodo *sig = cabeza->sig;
        cabeza->sig = prev;
        prev = cabeza;
        cabeza = sig;
    }
    return prev;
}
